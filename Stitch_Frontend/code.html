04  def geometric_qk_attention(query: torch.Tensor, key: torch.Tensor, value: torch.Tensor):
05      # Grant's intuition @ 25:12: Dot product measures vector projection resonance
06      d_k = query.size(-1)
07      
08      # [Batch, Heads, Seq_q, d_k] @ [Batch, Heads, d_k, Seq_k] -> [Batch, Heads, Seq_q, Seq_k]
09      scores = torch.matmul(query, key.transpose(-2, -1))
10      
11      # Scaling prevents high variance in large dimensional spaces (avoids vanishing softmax gradients)
12      scaled_scores = scores / (d_k ** 0.5)
13      
14      # Softmax normalizes resonance scores into dynamic distribution weights

19      return context, weights
